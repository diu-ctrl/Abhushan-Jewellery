## STEP 0.2 — Open Apps Script and paste the backend code

```javascript
/*************************************************************
 * ABHUSHAN — Orders & Appointments cloud store (Google Sheets)
 * Deployment: Web app | Execute as: Me | Who has access: Anyone
 * API (already wired in the website code):
 *   GET  ?action=orders&key=KEY        -> {ok, orders:[rows]}
 *   GET  ?action=appointments&key=KEY  -> {ok, appointments:[rows]}
 *   POST JSON {key, action:'order', order:{...}}
 *   POST JSON {key, action:'appointment', appointment:{...}}
 *************************************************************/
var ADMIN_KEY = 'AbhushanStudio_2026'; // must match cart.js + admin.html

var ORDER_HEADERS = ['id','placed_at','status','name','phone','email',
                     'address','city','pin','notes','items',
                     'subtotal','shipping','total','page'];
var APPT_HEADERS  = ['id','created_at','name','email','phone',
                     'date','time','service','notes','status'];

function ensureSheet_(name, headers) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.appendRow(headers);
    sh.setFrozenRows(1);
  }
  return sh;
}

function readRows_(name, headers) {
  var sh = ensureSheet_(name, headers);
  var values = sh.getDataRange().getValues();
  if (values.length < 2) return [];
  var out = [];
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][0]) === '') continue;
    var o = {};
    for (var c = 0; c < headers.length; c++) o[headers[c]] = values[i][c];
    out.push(o);
  }
  return out;
}

function doGet(e) {
  var p = (e && e.parameter) || {};
  if (String(p.key || '') !== ADMIN_KEY) return json_({ok:false, error:'bad-key'});
  if (p.action === 'orders')       return json_({ok:true, orders:       readRows_('Orders', ORDER_HEADERS)});
  if (p.action === 'appointments') return json_({ok:true, appointments: readRows_('Appointments', APPT_HEADERS)});
  return json_({ok:false, error:'unknown-action'});
}

function doPost(e) {
  var body = {};
  try { body = JSON.parse(e.postData.contents); }
  catch (err) { return json_({ok:false, error:'bad-json'}); }
  if (String(body.key || '') !== ADMIN_KEY) return json_({ok:false, error:'bad-key'});

  if (body.action === 'order' && body.order) {
    var o = body.order, c = o.customer || {};
    var sh = ensureSheet_('Orders', ORDER_HEADERS);
    sh.appendRow([
      String(o.id || ''), "'" + String(o.at || new Date().toISOString()),
      String(o.status || 'new'),
      String(c.name || ''), c.phone ? "'" + String(c.phone) : '', String(c.email || ''),
      String(c.address || ''), String(c.city || ''), c.pin ? "'" + String(c.pin) : '',
      String(c.notes || ''),
      itemsText_(o.items),
      Number(o.subtotal) || 0, Number(o.shipping) || 0, Number(o.total) || 0,
      String(o.page || '')
    ]);
    return json_({ok:true, saved:'order', id:String(o.id || '')});
  }

  if (body.action === 'appointment' && body.appointment) {
    var a = body.appointment;
    var sh2 = ensureSheet_('Appointments', APPT_HEADERS);
    sh2.appendRow([
      String(a.id || ''), "'" + String(a.created_at || new Date().toISOString()),
      String(a.name || ''), String(a.email || ''),
      a.phone ? "'" + String(a.phone) : '',
      String(a.date || ''), String(a.time || ''), String(a.service || ''),
      String(a.notes || ''), String(a.status || 'Pending')
    ]);
    return json_({ok:true, saved:'appointment', id:String(a.id || '')});
  }

  return json_({ok:false, error:'unknown-action'});
}

function itemsText_(items) {
  var arr = items || [], parts = [];
  for (var i = 0; i < arr.length; i++) {
    var s = String(arr[i].name || '');
    if (arr[i].size)  s += ' (Size ' + arr[i].size + ')';
    if (arr[i].price) s += ' \u2014 \u20B9' + arr[i].price;
    parts.push(s);
  }
  return parts.join(' | ');
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
                       .setMimeType(ContentService.MimeType.JSON);
}
```

3. Press **Ctrl+S** to save. (The leading `'` trick keeps phone numbers and ISO

## STEP 3 — `script.js`: FIX the blurred screen after closing the menu

Root cause: `closeMobileMenu()` never removes the `.active` class from `#overlay-bg`
(`position:fixed; inset:0; background:rgba(26,23,20,.6); backdrop-filter:blur(4px);
pointer-events:all`). Once the menu is closed, that overlay stays on screen forever —
the page looks blurred/dark and every tap lands on the overlay.

Find (lines 388–399):
```js
  function closeMobileMenu() {
    hamburger.classList.remove('open');
    hamburger.setAttribute('aria-expanded', 'false');
    mobileMenu.classList.remove('open');
    header.classList.remove('menu-open');
    document.body.style.overflow = '';
    setTimeout(() => {
      if (!mobileMenu.classList.contains('open')) {
        // leave display as flex, CSS handles visibility
      }
    }, 300);
  }
```

Replace with:
```js
  function closeMobileMenu() {
    hamburger.classList.remove('open');
    hamburger.setAttribute('aria-expanded', 'false');
    mobileMenu.classList.remove('open');
    header.classList.remove('menu-open');
    overlayBg.classList.remove('active'); /* PHASE 6 FIX — clear dark blur overlay */
    document.body.style.overflow = '';
    setTimeout(() => {
      if (!mobileMenu.classList.contains('open')) {
        mobileMenu.style.display = ''; /* PHASE 6 FIX — fully remove from layout */
      }
    }, 300);
  }
```

This single function is called by EVERY close path (hamburger, backdrop click,
Escape key, nav-link taps) — so one fix covers all of them.
